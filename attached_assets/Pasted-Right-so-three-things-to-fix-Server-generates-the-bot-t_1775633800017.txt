One small fix needed in server.js — the download route uses a query param for auth instead of a header, so update the auth middleware to accept both:
javascriptconst auth = (req, res, next) => {
  let encoded = null;

  const authHeader = req.headers.authorization;
  if (authHeader) {
    encoded = authHeader.split(' ')[1];
  } else if (req.query.auth) {
    encoded = req.query.auth;
  }

  if (!encoded) return res.status(401).json({ error: 'Unauthorised' });

  const [, pwd] = Buffer.from(encoded, 'base64').toString().split(':');
  if (pwd === process.env.DASHBOARD_PASSWORD) {
    next();
  } else {
    res.status(401).json({ error: 'Invalid password' });
  }
};
