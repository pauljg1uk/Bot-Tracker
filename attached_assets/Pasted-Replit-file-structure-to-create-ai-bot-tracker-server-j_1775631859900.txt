javascriptrequire('dotenv').config();
const express = require('express');
const { Pool } = require('pg');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());
app.use(express.static('public'));

const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  ssl: { rejectUnauthorized: false }

// --- Auth middleware for dashboard API routes ---
const auth = (req, res, next) => {
  const authHeader = req.headers.authorization;
  if (!authHeader) {
    return res.status(401).json({ error: 'Unauthorised' });
  }
  const base64 = authHeader.split(' ')[1];
  const [, password] = Buffer.from(base64, 'base64').toString().split(':');
  if (password === process.env.DASHBOARD_PASSWORD) {
    next();
  } else {
    res.status(401).json({ error: 'Invalid password' });
  }
};

// --- Receive bot hit from Cloudflare Worker ---
app.post('/api/hit', async (req, res) => {
  const { api_key, url, bot_name, user_agent, status_code, country, referrer } = req.body;
  try {
    const clientResult = await pool.query(
      'SELECT id FROM clients WHERE api_key = $1', [api_key]
    );
    if (clientResult.rows.length === 0) {
      return res.status(401).json({ error: 'Invalid API key' });
    }
    const client_id = clientResult.rows[0].id;
    await pool.query(
      `INSERT INTO bot_hits (client_id, url, bot_name, user_agent, status_code, country, referrer)
       VALUES ($1, $2, $3, $4, $5, $6, $7)`,
      [client_id, url, bot_name, user_agent, status_code, country, referrer]
    );
    res.json({ success: true });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Server error' });
  }
});

// --- Get all clients ---
app.get('/api/clients', auth, async (req, res) => {
  try {
    const result = await pool.query(
      'SELECT id, name, domain, api_key, created_at FROM clients ORDER BY created_at DESC'
    );
    res.json(result.rows);
  } catch (err) {
    res.status(500).json({ error: 'Server error' });
  }
});

// --- Get stats for a client ---
app.get('/api/stats/:clientId', auth, async (req, res) => {
  const { clientId } = req.params;
  const days = parseInt(req.query.days) || 30;
  try {
    const [totalHits, byBot, topPages, overTime] = await Promise.all([
      pool.query(
        `SELECT COUNT(*) as total FROM bot_hits 
         WHERE client_id = $1 AND timestamp > NOW() - INTERVAL '${days} days'`,
        [clientId]
      ),
      pool.query(
        `SELECT bot_name, COUNT(*) as hits FROM bot_hits 
         WHERE client_id = $1 AND timestamp > NOW() - INTERVAL '${days} days'
         GROUP BY bot_name ORDER BY hits DESC`,
        [clientId]
      ),
      pool.query(
        `SELECT url, COUNT(*) as hits FROM bot_hits 
         WHERE client_id = $1 AND timestamp > NOW() - INTERVAL '${days} days'
         GROUP BY url ORDER BY hits DESC LIMIT 10`,
        [clientId]
      ),
      pool.query(
        `SELECT DATE(timestamp) as date, COUNT(*) as hits FROM bot_hits 
         WHERE client_id = $1 AND timestamp > NOW() - INTERVAL '${days} days'
         GROUP BY DATE(timestamp) ORDER BY date ASC`,
        [clientId]
      )
    ]);
    res.json({
      total: totalHits.rows[0].total,
      byBot: byBot.rows,
      topPages: topPages.rows,
      overTime: overTime.rows
    });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Server error' });
  }
});

// --- Recent hits for a client ---
app.get('/api/hits/:clientId', auth, async (req, res) => {

    const result = await pool.query(

       ORDER BY timestamp DESC LIMIT 200`,
      [clientId]
    );
    res.json(result.rows);
  } catch (err) {
    res.status(500).json({ error: 'Server error' });
  }
});

<script>
  let password = '';

  function login() {
    password = document.getElementById('password-input').value;
    fetch('/api/clients', {
      headers: { 'Authorization': 'Basic ' + btoa('admin:' + password) }
    }).then(r => {
      if (r.ok) {
        document.getElementById('login-screen').style.display = 'none';
        document.getElementById('app').style.display = 'block';
        loadClients();
      } else {
        document.getElementById('login-error').style.display = 'block';
      }
    });
  }

  function authHeaders() {
    return { 'Authorization': 'Basic ' + btoa('admin:' + password), 'Content-Type': 'application/json' };
  }

  function loadClients() {
    fetch('/api/clients', { headers: authHeaders() })
      .then(r => r.json())
      .then(clients => {
        const select = document.getElementById('client-select');
        select.innerHTML = '<option value="">Select client...</option>';
        clients.forEach(c => {
          select.innerHTML += `<option value="${c.id}">${c.name} — ${c.domain}</option>`;
        });
      });
  }

  function loadClient() {
    const clientId = document.getElementById('client-select').value;
    const days = document.getElementById('days-select').value;
    if (!clientId) {
      document.getElementById('dashboard').style.display = 'none';
      document.getElementById('no-client').style.display = 'block';
      return;
    }
    document.getElementById('dashboard').style.display = 'block';
    document.getElementById('no-client').style.display = 'none';

    Promise.all([
      fetch(`/api/stats/${clientId}?days=${days}`, { headers: authHeaders() }).then(r => r.json()),
      fetch(`/api/hits/${clientId}?days=${days}`, { headers: authHeaders() }).then(r => r.json())
    ]).then(([stats, hits]) => {
      document.getElementById('stat-total').textContent = stats.total;
      document.getElementById('stat-bots').textContent = stats.byBot.length;
      document.getElementById('stat-pages').textContent = stats.topPages.length;

      document.getElementById('bot-breakdown').innerHTML = stats.byBot.length
        ? stats.byBot.map(b => `
          <div class="bot-row">
            <span class="bot-badge">${b.bot_name}</span>
            <span class="hits-count">${b.hits} hits</span>
          </div>`).join('')
        : '<div class="empty">No data yet</div>';

      document.getElementById('top-pages').innerHTML = stats.topPages.length
        ? stats.topPages.map(p => `
          <div class="bot-row">
            <span class="url-text" title="${p.url}">${p.url}</span>
            <span class="hits-count">${p.hits}</span>
          </div>`).join('')
        : '<div class="empty">No data yet</div>';

      document.getElementById('hits-table').innerHTML = hits.length
        ? hits.map(h => `
          <tr>
            <td>${new Date(h.timestamp).toLocaleString()}</td>
            <td><span class="bot-badge">${h.bot_name}</span></td>
            <td class="mono" title="${h.url}">${h.url}</td>
            <td>${h.country || '—'}</td>
            <td>${h.status_code || '—'}</td>
          </tr>`).join('')
        : '<tr><td colspan="5" style="color:#64748b;padding:20px 0">No hits yet</td></tr>';
    });
  }

javascriptconst AI_BOTS = [
  { name: 'GPTBot', pattern: 'GPTBot' },
  { name: 'ChatGPT-User', pattern: 'ChatGPT-User' },
  { name: 'ClaudeBot', pattern: 'ClaudeBot' },
  { name: 'Anthropic', pattern: 'anthropic-ai' },
  { name: 'Google-Extended', pattern: 'Google-Extended' },
  { name: 'PerplexityBot', pattern: 'PerplexityBot' },
  { name: 'Bytespider', pattern: 'Bytespider' },
  { name: 'CCBot', pattern: 'CCBot' },
  { name: 'Meta-ExternalAgent', pattern: 'Meta-ExternalAgent' },
  { name: 'Cohere', pattern: 'cohere-ai' },
  { name: 'YouBot', pattern: 'YouBot' },
  { name: 'Diffbot', pattern: 'Diffbot' },
  { name: 'Applebot-Extended', pattern: 'Applebot-Extended' },
];

const API_ENDPOINT = 'https://YOUR-REPLIT-URL.repl.co/api/hit';
const CLIENT_API_KEY = 'YOUR-CLIENT-API-KEY';

addEventListener('fetch', event => {
  event.respondWith(handleRequest(event.request, event));
});

async function handleRequest(request, event) {
  const response = await fetch(request);
  const userAgent = request.headers.get('User-Agent') || '';
  const matchedBot = AI_BOTS.find(bot => userAgent.includes(bot.pattern));

  if (matchedBot) {
    const url = new URL(request.url);
    event.waitUntil(
      fetch(API_ENDPOINT, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          api_key: CLIENT_API_KEY,
          url: url.pathname,
          bot_name: matchedBot.name,
          user_agent: userAgent,
          status_code: response.status,
          country: request.cf ? request.cf.country : null,
          referrer: request.headers.get('Referer') || null
        })
      })
    );
  }

  return response;
}
